cmake_minimum_required(VERSION 3.16)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(zvec_collection_benchmark LANGUAGES CXX)
endif()

if(IOS)
  add_executable(zvec_collection_benchmark MACOSX_BUNDLE benchmark.cc)
  set_target_properties(zvec_collection_benchmark PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${PROJECT_ROOT_DIR}/cmake/iOSBundleInfo.plist.in")
else()
  add_executable(zvec_collection_benchmark benchmark.cc)
endif()

target_compile_features(zvec_collection_benchmark PRIVATE cxx_std_17)
if(TARGET zvec::static)
  target_link_libraries(zvec_collection_benchmark PRIVATE zvec::static)
elseif(TARGET zvec_shared)
  target_link_libraries(zvec_collection_benchmark PRIVATE zvec_shared)
elseif(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  set(ZVEC_SDK_ROOT "" CACHE PATH "Root of an extracted Zvec C++ SDK")
  find_path(ZVEC_INCLUDE_DIR zvec/db/collection.h HINTS "${ZVEC_SDK_ROOT}/include")
  find_library(ZVEC_COLLECTION_LIBRARY NAMES zvec zvec_shared
               HINTS "${ZVEC_SDK_ROOT}/lib")
  if(NOT ZVEC_INCLUDE_DIR OR NOT ZVEC_COLLECTION_LIBRARY)
    message(FATAL_ERROR "Set ZVEC_SDK_ROOT or provide ZVEC_INCLUDE_DIR and ZVEC_COLLECTION_LIBRARY")
  endif()
  target_include_directories(zvec_collection_benchmark PRIVATE "${ZVEC_INCLUDE_DIR}")
  target_link_libraries(zvec_collection_benchmark PRIVATE "${ZVEC_COLLECTION_LIBRARY}")
else()
  message(FATAL_ERROR "The Collection benchmark requires the public SDK (BUILD_ZVEC_SHARED=ON on desktop)")
endif()

if(WIN32)
  target_link_libraries(zvec_collection_benchmark PRIVATE Psapi)
endif()
