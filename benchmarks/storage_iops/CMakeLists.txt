cmake_minimum_required(VERSION 3.20)
project(zvec_storage_iops LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Threads REQUIRED)
add_executable(storage_iops probe.cc)
target_link_libraries(storage_iops PRIVATE Threads::Threads)
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_link_libraries(storage_iops PRIVATE proc)
endif()
if(MSVC)
  target_compile_options(storage_iops PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(storage_iops PRIVATE -Wall -Wextra -Werror)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(storage_iops PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
endif()
